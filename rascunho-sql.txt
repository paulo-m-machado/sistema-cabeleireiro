2. Tabelas Base

 (Entidades Principais)CREATE TABLE clientes (
    id INT AUTO_INCREMENT PRIMARY KEY,
    nome TEXT,
    endereco TEXT,
    contato TEXT,
    data_nascimento DATE,
    preferencia_1 BOOLEAN,
    preferencia_2 BOOLEAN,
    descricao TEXT
);

CREATE TABLE funcionarios (
    id INT AUTO_INCREMENT PRIMARY KEY,
    nome TEXT,
    endereco TEXT,
    contato TEXT,
    cpf TEXT,
    data_nascimento DATE,
    funcao TEXT,
    disponibilidade TEXT
);

CREATE TABLE fornecedores (
    id INT AUTO_INCREMENT PRIMARY KEY,
    cnpj TEXT,
    nome_fantasia TEXT,
    descricao TEXT,
    data_cadastro DATE,
    status BOOLEAN,
    data_cancelamento DATE,
    usuario_cadastrou INT,
    FOREIGN KEY (usuario_cadastrou) REFERENCES funcionarios(id)
);

CREATE TABLE produtos (
    id INT AUTO_INCREMENT PRIMARY KEY,
    descricao TEXT,
    marca TEXT,
    qtd_estoque FLOAT,
    qtd_minima FLOAT,
    qtd_maxima FLOAT,
    vlr_venda FLOAT,
    data_inclusao DATE,
    data_ultima_venda DATE,
    data_alteracao DATE,
    usuario_alterou INT,
    status BOOLEAN,
    data_cancelamento DATE,
    usuario_cadastrou INT,
    FOREIGN KEY (usuario_cadastrou) REFERENCES funcionarios(id),
    FOREIGN KEY (usuario_alterou) REFERENCES funcionarios(id)
);


3. Tabelas de Transação (Vendas, Compras e Atendimentos)
CREATE TABLE atendimentos (
    id INT AUTO_INCREMENT PRIMARY KEY,
    funcionario_id INT,
    cliente_id INT,
    nome_cliente INT, -- No diagrama está como INT, mas geralmente seria TEXT ou FK
    horario TIMESTAMP,
    duracao TIMESTAMP,
    servico TEXT,
    foco BOOLEAN,
    FOREIGN KEY (funcionario_id) REFERENCES funcionarios(id),
    FOREIGN KEY (cliente_id) REFERENCES clientes(id)
);

CREATE TABLE vendas (
    id INT AUTO_INCREMENT PRIMARY KEY,
    cliente_id INT,
    funcionario_id INT,
    qtd_produto FLOAT,
    valor FLOAT,
    FOREIGN KEY (cliente_id) REFERENCES clientes(id),
    FOREIGN KEY (funcionario_id) REFERENCES funcionarios(id)
);

CREATE TABLE compras (
    id INT AUTO_INCREMENT PRIMARY KEY,
    fornecedor_id INT,
    valor FLOAT,
    quantidade FLOAT,
    data_compra DATE,
    data_entrega DATE,
    status TEXT,
    FOREIGN KEY (fornecedor_id) REFERENCES fornecedores(id)
);

4. Tabelas Associativas (Relacionamentos N:N)

CREATE TABLE atendimento_produto (
    id INT AUTO_INCREMENT PRIMARY KEY,
    atendimento_id INT,
    produtos_id INT,
    FOREIGN KEY (atendimento_id) REFERENCES atendimentos(id),
    FOREIGN KEY (produtos_id) REFERENCES produtos(id)
);

CREATE TABLE venda_produto (
    id INT AUTO_INCREMENT PRIMARY KEY,
    venda_id INT,
    produto_id INT,
    FOREIGN KEY (venda_id) REFERENCES vendas(id),
    FOREIGN KEY (produto_id) REFERENCES produtos(id)
);

CREATE TABLE compra_produto (
    id INT AUTO_INCREMENT PRIMARY KEY,
    compra_id INT,
    produto_id INT,
    FOREIGN KEY (compra_id) REFERENCES compras(id),
    FOREIGN KEY (produto_id) REFERENCES produtos(id)
);

Observações Importantes:
Tipos de Dados: No seu diagrama, muitos campos estão como TEXT(65535). No MySQL, você pode usar apenas TEXT ou definir um limite como VARCHAR(255) para economizar espaço se os nomes forem curtos.

Campos de Data: Usei DATE para nascimentos e cadastros, e TIMESTAMP para horários de atendimento conforme indicado.

FKs: Note que a tabela produtos e fornecedores possuem chaves estrangeiras ligadas a funcionarios (quem cadastrou/alterou), por isso a tabela de funcionários deve ser criada primeiro.
